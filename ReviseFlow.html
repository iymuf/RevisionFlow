// ────────────────────────────────────── // BLOCK REPETITION HELPER // Returns the next calendar block date for a subject, // skipping (blockInterval - 1) blocks after the first block that follows startDate. // ────────────────────────────────────── const getNextBlockDate = (subjectId, startDate, blockInterval, calendarBlocks) => { if (!blockInterval || blockInterval < 1) return null; if (!calendarBlocks || !calendarBlocks[subjectId]) return null; const dates = calendarBlocks[subjectId].dates; if (!dates || dates.length === 0) return null; // Find the first block that is strictly after startDate let baseIdx = dates.findIndex(d => d > startDate); if (baseIdx === -1) { // All blocks are before startDate → use the last one as the base baseIdx = dates.length - 1; } // The target block is blockInterval blocks after the base (1-indexed) const targetIdx = baseIdx + (blockInterval - 1); if (targetIdx >= dates.length) { // If not enough blocks exist, fall back to the last available date return dates[dates.length - 1]; } return dates[targetIdx]; };
